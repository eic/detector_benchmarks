    BEFORE:
      shell: """
      echo "Merging simulation files into: {output}"
      hadd -f {output.sim_hadd} {input.sim}
      OUTPUT_PREFIX="$(dirname {output.hitsxy_png})"
      echo "Generating hit maps..."
      root -l -b -q {input.script_hitsmap}'("{output.sim_hadd}", ...)'
      root -l -b -q {input.script_nhits_eta}'("{output.sim_hadd}", ...)'
      """
    
    AFTER:
      shell: """
      OUTPUT_PREFIX="$(dirname {output.hitsxy_png})"
      mkdir -p $OUTPUT_PREFIX
      CONFIG_FILE="$OUTPUT_PREFIX/.config_sim_files.json"
      # Create JSON with file list
      echo '{{' > $CONFIG_FILE
      echo '  "sim_files": [' >> $CONFIG_FILE
      FILES=({input.sim})
      for ((i=0; i<${{#FILES[@]}}; i++)); do
        if ((i < ${{#FILES[@]}} - 1)); then
          echo "    \"${{FILES[i]}}\"," >> $CONFIG_FILE
        else
          echo "    \"${{FILES[i]}}\"" >> $CONFIG_FILE
        fi
      done
      echo '  ]' >> $CONFIG_FILE
      echo '}}' >> $CONFIG_FILE
      # Run analysis scripts with config
      echo "Generating hit maps and Nhits vs eta for files: {input.sim}"
      root -l -b -q {input.script_hitsmap}'("'$CONFIG_FILE'", "'$OUTPUT_PREFIX'")'
      root -l -b -q {input.script_nhits_eta}'("'$CONFIG_FILE'", "{params.MOMENTUM}", "'$OUTPUT_PREFIX'")'
      rm -f $CONFIG_FILE
      """

JSON Configuration Format

    "sim_output/tracking_performance/epic_craterlake_tracking_only/pi-/1GeV/45to135deg/pi-_1GeV_45to135deg.0000.edm4hep.root",
    "sim_output/tracking_performance/epic_craterlake_tracking_only/pi-/1GeV/130to177deg/pi-_1GeV_130to177deg.0000.edm4hep.root"
  ]
}

TChain Integration Pattern
───────────────────────────────────────────────────────────────────────────

File I/O Architecture:

  Input Files (3 EDM4HEP ROOT files)
           ↓
  TChain Creation with file list
           ↓
  TChain::Add() for each file
           ↓
  Analysis code (Tree::Draw or TTreeReader)
           ↓
  Output plots (PNG, EPS, ROOT)

Backward Compatibility
───────────────────────────────────────────────────────────────────────────

The scripts support two input formats:

1. JSON Config (NEW):
   root -l -b -q draw_hits.C'("/path/to/config.json", "/output/dir")'

2. Single File (LEGACY):
   root -l -b -q draw_hits.C'("/path/to/file.root", "/output/dir")'

Detection mechanism:
  if (config_json.Contains(".json")) {
    // Parse JSON and create TChain
  } else {
    // Treat as single file path (legacy)
  }
